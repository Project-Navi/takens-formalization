import Mathlib

/-!
# Sard's Theorem

Sard's theorem states that the set of critical values of a sufficiently smooth map
between finite-dimensional real vector spaces has Lebesgue measure zero.

A point `x` is a **critical point** of `f` if the Fréchet derivative `fderiv ℝ f x`
is not surjective. The image of the set of critical points is the set of
**critical values**.

## Main results

* `sard_equidimensional`: **Sard's theorem** for C¹ maps `f : E → E` (equidimensional
  case), proved using the area formula / Jacobian determinant.

* `addHaar_image_eq_zero_of_finrank_lt`: The image of any set under a C¹ map from a
  lower-dimensional space has measure zero (Hausdorff dimension argument).

* `sard_of_finrank_eq`: Sard's theorem for C¹ maps between spaces of equal dimension,
  reduced to the equidimensional case via a continuous linear equivalence.

* `sard`: The full **Sard's theorem** for C^∞ maps between arbitrary finite-dimensional
  real normed spaces (modulo `sard_of_finrank_gt` for the case `finrank E > finrank F`).

## Proof strategy

The proof splits into three cases based on `finrank ℝ E` vs `finrank ℝ F`:

1. **finrank E < finrank F**: The image of `f` has measure zero because Lipschitz maps
   preserve Hausdorff dimension, and `dimH(E) < finrank F`.

2. **finrank E = finrank F**: At critical points, `det(Df) = 0`. The area formula
   (Jacobian formula) gives `μ(f '' C) ≤ ∫_C |det Df| dμ = 0`.

3. **finrank E > finrank F**: The standard proof uses strong induction on `finrank E`,
   the implicit function theorem to reduce dimension, and Taylor's theorem for the
   "deep" critical set. This case remains as `sorry`.

## References

* J. Milnor, *Topology from the Differentiable Viewpoint*, 1965.
* V. Guillemin and A. Pollack, *Differential Topology*, 1974.
-/

open MeasureTheory Set Function ENNReal Topology Filter FiniteDimensional

/-! ### Equidimensional case: maps E → E -/

section Equidimensional

variable {E : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E] [FiniteDimensional ℝ E]
  [MeasurableSpace E] [BorelSpace E]

/-- At a critical point of a map `E → E` (where the derivative is not surjective),
the determinant of the derivative vanishes. -/
lemma det_fderiv_eq_zero_of_not_surjective (L : E →L[ℝ] E)
    (h : ¬ Surjective L) : L.det = 0 := by
  contrapose! h
  exact (LinearMap.equivOfDetNeZero (L : E →ₗ[ℝ] E) h).surjective

/-- The set of critical points of a C¹ map `E → E` is closed. -/
lemma isClosed_criticalSet (f : E → E) (hf : ContDiff ℝ 1 f) :
    IsClosed {x : E | ¬ Surjective (fderiv ℝ f x)} := by
  have h_preimage : {x : E | ¬Surjective (fderiv ℝ f x)} =
      {x : E | (fderiv ℝ f x).det = 0} := by
    ext x
    simp [det_fderiv_eq_zero_of_not_surjective]
    constructor <;> intro h
    · grind +suggestions
    · intro h_surj
      have := LinearMap.isUnit_det (fderiv ℝ f x |> ContinuousLinearMap.toLinearMap)
      simp_all +decide [LinearMap.isUnit_iff_ker_eq_bot]
      exact this (LinearMap.ker_eq_bot_of_injective
        (LinearMap.injective_iff_surjective.mpr h_surj))
  convert isClosed_eq ?_ ?_
  all_goals try infer_instance
  · exact ContinuousLinearMap.continuous_det.comp (hf.continuous_fderiv one_ne_zero)
  · fun_prop

/-- **Sard's Theorem** (equidimensional case): For a C¹ map `f : E → E` between
finite-dimensional real normed spaces, the image of the set of critical points
(where the derivative is not surjective) has measure zero.

The proof uses the area formula (Jacobian change-of-variables):
  `μ(f '' C) ≤ ∫_C |det(Df)| dμ = 0`
since `det(Df(x)) = 0` at every critical point `x ∈ C`. -/
theorem sard_equidimensional (μ : Measure E) [μ.IsAddHaarMeasure]
    (f : E → E) (hf : ContDiff ℝ 1 f) :
    μ (f '' {x | ¬ Surjective (fderiv ℝ f x)}) = 0 := by
  have hC : MeasurableSet {x | ¬Surjective (fderiv ℝ f x)} :=
    (isClosed_criticalSet f hf).measurableSet
  have h_area : μ (f '' {x | ¬Surjective (fderiv ℝ f x)}) ≤
      ∫⁻ x in {x | ¬Surjective (fderiv ℝ f x)}, ENNReal.ofReal |(fderiv ℝ f x).det| ∂μ :=
    addHaar_image_le_lintegral_abs_det_fderiv μ hC
      (fun x _ => (hf.contDiffAt.differentiableAt (by norm_num)).hasFDerivAt.hasFDerivWithinAt)
  refine le_antisymm (h_area.trans ?_) bot_le
  rw [lintegral_congr_ae, lintegral_zero]
  filter_upwards [ae_restrict_mem hC] with x hx
  simp [det_fderiv_eq_zero_of_not_surjective _ hx]

end Equidimensional

/-! ### Low-dimensional case: finrank E < finrank F -/

section LowDimensional

variable {E F : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E] [FiniteDimensional ℝ E]
  [NormedAddCommGroup F] [NormedSpace ℝ F] [FiniteDimensional ℝ F]
  [MeasurableSpace E] [BorelSpace E]
  [MeasurableSpace F] [BorelSpace F]

/-- The Hausdorff dimension of any subset of a finite-dimensional space
is at most the finrank. -/
lemma dimH_le_finrank (s : Set E) :
    dimH s ≤ Module.finrank ℝ E :=
  (dimH_mono (subset_univ s)).trans (Real.dimH_univ_eq_finrank E).le

/-- A C¹ map on a closed ball in a finite-dimensional space is Lipschitz. -/
lemma lipschitzOnWith_closedBall_of_contDiff
    (f : E → F) (hf : ContDiff ℝ 1 f) (x₀ : E) (r : ℝ) :
    ∃ K : NNReal, LipschitzOnWith K f (Metric.closedBall x₀ r) := by
  obtain ⟨K, hK⟩ : ∃ K : NNReal, ∀ x ∈ Metric.closedBall x₀ r, ‖fderiv ℝ f x‖₊ ≤ K := by
    have : ContinuousOn (fun x => ‖fderiv ℝ f x‖₊) (Metric.closedBall x₀ r) := by fun_prop
    exact ⟨_, fun x hx => le_csSup
      ((isCompact_closedBall x₀ r).image_of_continuousOn this).bddAbove
      (mem_image_of_mem _ hx)⟩
  exact ⟨K, Convex.lipschitzOnWith_of_nnnorm_hasFDerivWithin_le
    (fun x _ => (hf.contDiffAt.differentiableAt (by norm_num)).hasFDerivAt.hasFDerivWithinAt)
    hK (convex_closedBall _ _)⟩

/-
PROBLEM
A C¹ image of a set in a lower-dimensional space has measure zero.
This is the "easy case" of Sard's theorem, using Hausdorff dimension.

PROVIDED SOLUTION
Cover s by countably many closed balls: s ⊆ ⋃ n, s ∩ closedBall 0 n. For each n, by lipschitzOnWith_closedBall_of_contDiff, f is K-Lipschitz on closedBall 0 n, hence on s ∩ closedBall 0 n. By LipschitzOnWith.dimH_image_le, dimH(f '' (s ∩ closedBall 0 n)) ≤ dimH(s ∩ closedBall 0 n) ≤ finrank ℝ E (by dimH_le_finrank). Since finrank ℝ E < finrank ℝ F, we have dimH < finrank ℝ F cast as NNReal. Use measure_zero_of_dimH_lt with μ ≪ μH[finrank ℝ F] (via Measure.absolutelyContinuous_isAddHaarMeasure). Each piece has measure zero. Then f '' s ⊆ ⋃ n, f '' (s ∩ closedBall 0 n) so by measure_iUnion_null we are done.
-/
theorem addHaar_image_eq_zero_of_finrank_lt
    (μ : Measure F) [μ.IsAddHaarMeasure]
    (f : E → F) (hf : ContDiff ℝ 1 f) (s : Set E)
    (hdim : Module.finrank ℝ E < Module.finrank ℝ F) :
    μ (f '' s) = 0 := by
  -- Cover s by countably many closed balls: s ⊆ ⋃ n, s ∩ closedBall 0 n. For each n, by lipschitzOnWith_closedBall_of_contDiff, f is K-Lipschitz on closedBall 0 n, hence on s ∩ closedBall 0 n. By LipschitzOnWith.dimH_image_le, dimH(f '' (s ∩ closedBall 0 n)) ≤ dimH(s ∩ closedBall 0 n) ≤ finrank ℝ E (by dimH_le_finrank).
  have h_cover : ∀ n : ℕ, μ (f '' (s ∩ Metric.closedBall 0 (n : ℝ))) = 0 := by
    intro n
    have h_lip : ∃ K : NNReal, LipschitzOnWith K f (Metric.closedBall 0 (n : ℝ)) := by
      exact?
    obtain ⟨K, hK⟩ := h_lip
    have h_dimH : dimH (f '' (s ∩ Metric.closedBall 0 (n : ℝ))) ≤ dimH (s ∩ Metric.closedBall 0 (n : ℝ)) := by
      apply_rules [ LipschitzOnWith.dimH_image_le ];
      exact hK.mono ( Set.inter_subset_right )
    have h_finrank : dimH (s ∩ Metric.closedBall 0 (n : ℝ)) ≤ Module.finrank ℝ E := by
      exact?
    have h_lt : dimH (f '' (s ∩ Metric.closedBall 0 (n : ℝ))) < Module.finrank ℝ F := by
      exact lt_of_le_of_lt h_dimH ( lt_of_le_of_lt h_finrank ( mod_cast hdim ) )
    have h_zero : μ (f '' (s ∩ Metric.closedBall 0 (n : ℝ))) = 0 := by
      apply_rules [ measure_zero_of_dimH_lt ];
      have h_abs_cont : ∀ (μ : Measure F) [μ.IsAddHaarMeasure], μ ≪ μH[↑↑(Module.finrank ℝ F)] := by
        exact?;
      exact h_abs_cont μ
    exact h_zero;
  rw [ show f '' s = ⋃ n : ℕ, f '' ( s ∩ Metric.closedBall 0 n ) from ?_ ];
  · exact MeasureTheory.measure_iUnion_null h_cover;
  · ext x
    simp [Set.mem_image, Set.mem_iUnion];
    exact ⟨ fun ⟨ y, hy, hxy ⟩ => ⟨ ⌈‖y‖⌉₊, y, ⟨ hy, Nat.le_ceil _ ⟩, hxy ⟩, fun ⟨ n, y, ⟨ hy, hn ⟩, hxy ⟩ => ⟨ y, hy, hxy ⟩ ⟩

/-- The range of a C¹ map from a lower-dimensional space has measure zero. -/
theorem addHaar_range_eq_zero_of_finrank_lt
    (μ : Measure F) [μ.IsAddHaarMeasure]
    (f : E → F) (hf : ContDiff ℝ 1 f)
    (hdim : Module.finrank ℝ E < Module.finrank ℝ F) :
    μ (range f) = 0 := by
  rw [← image_univ]
  exact addHaar_image_eq_zero_of_finrank_lt μ f hf univ hdim

end LowDimensional

/-! ### Reduction from general to equidimensional case -/

section Reduction

variable {E F : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E] [FiniteDimensional ℝ E]
  [NormedAddCommGroup F] [NormedSpace ℝ F] [FiniteDimensional ℝ F]
  [MeasurableSpace E] [BorelSpace E]
  [MeasurableSpace F] [BorelSpace F]

/-- The critical set is preserved by precomposition with a continuous linear equivalence. -/
lemma criticalSet_comp_linearEquiv (f : E → F) (L : F ≃L[ℝ] E) :
    {x | ¬ Surjective (fderiv ℝ (L ∘ f) x)} =
    {x | ¬ Surjective (fderiv ℝ f x)} := by
  have h_chain : ∀ x, fderiv ℝ (L ∘ f) x = L.toContinuousLinearMap.comp (fderiv ℝ f x) := by
    intro x
    by_cases h : DifferentiableAt ℝ f x
    · exact (L.toContinuousLinearMap.hasFDerivAt.comp x h.hasFDerivAt).fderiv
    · rw [fderiv_zero_of_not_differentiableAt]
      · rw [fderiv_zero_of_not_differentiableAt h]; ext; simp +decide
      · grind +suggestions
  ext x; simp [h_chain]

/-
PROBLEM
**Sard's theorem** for maps between spaces of equal finite dimension.

PROVIDED SOLUTION
Let L := (ContinuousLinearEquiv.ofFinrankEq hdim).symm : F ≃L[ℝ] E. Let g := L ∘ f : E → E. Then g is C^1. The critical sets agree by criticalSet_comp_linearEquiv. Consider ν := Measure.map L μ which is addHaar on E by ContinuousLinearEquiv.isAddHaarMeasure_map. By sard_equidimensional, ν(g '' C_g) = 0. We need μ(f '' C_f) = 0. Note f '' C_f = L ⁻¹' (g '' C_g) because f = L.symm ∘ g. So μ(f '' C_f) = μ(L ⁻¹' (g '' C_g)) = (Measure.map L μ)(g '' C_g) = ν(g '' C_g) = 0, using Measure.map_apply.
-/
theorem sard_of_finrank_eq
    (μ : Measure F) [μ.IsAddHaarMeasure]
    (f : E → F) (hf : ContDiff ℝ 1 f)
    (hdim : Module.finrank ℝ E = Module.finrank ℝ F) :
    μ (f '' {x | ¬ Surjective (fderiv ℝ f x)}) = 0 := by
  obtain ⟨L, hL⟩ : ∃ L : F ≃L[ℝ] E, True := by
    refine' ⟨ _, trivial ⟩
    generalize_proofs at *; (
    exact ( LinearEquiv.toContinuousLinearEquiv ( LinearEquiv.ofFinrankEq _ _ <| by simp +decide [ hdim ] ) ) |> ContinuousLinearEquiv.symm |> ContinuousLinearEquiv.trans <| ContinuousLinearEquiv.refl ℝ E;) -- To preserve the `generalize_proofs` context. (This line can be removed after conversion to `generalize_proofs at *` at `all_goals`.)
  generalize_proofs at *; (
  -- Apply Sard's theorem to the map $L \circ f : E \to E$.
  have h_sard : (Measure.map L μ) ((L ∘ f) '' { x | ¬ Surjective (fderiv ℝ (L ∘ f) x) }) = 0 := by
    apply sard_equidimensional (Measure.map L μ) (L ∘ f) (by
    exact L.contDiff.comp hf)
  generalize_proofs at *; (
  -- Since $L$ is a linear equivalence, we have $L '' (f '' C_f) = (L ∘ f) '' C_f$.
  have h_image : L '' (f '' { x | ¬ Surjective (fderiv ℝ f x) }) = (L ∘ f) '' { x | ¬ Surjective (fderiv ℝ (L ∘ f) x) } := by
    simp +decide [ Set.ext_iff, Function.comp ];
    intro x; constructor <;> rintro ⟨ y, hy, rfl ⟩ <;> use y <;> simp_all +decide [ Function.comp, fderiv_comp ] ;
    · rw [ fderiv_comp ] <;> norm_num [ hf.contDiffAt.differentiableAt ];
      · rw [ show fderiv ℝ ( L : F → E ) ( f y ) = L.toContinuousLinearMap from by exact HasFDerivAt.fderiv ( L.hasFDerivAt ) ] ; aesop;
      · exact L.differentiableAt;
    · intro h; contrapose! hy; simp_all +decide [ Function.Surjective ] ;
      intro b; obtain ⟨ a, ha ⟩ := h ( L.symm b ) ; use a; erw [ fderiv_comp ] <;> norm_num [ ha, hf.contDiffAt.differentiableAt ] ;
      · erw [ L.fderiv ] ; aesop;
      · exact L.differentiableAt
  generalize_proofs at *; (
  rw [ Measure.map_apply ] at h_sard;
  · convert h_sard using 1 ; rw [ ← h_image ] ; simp +decide [ Set.preimage ] ;
    rfl;
  · exact L.continuous.measurable;
  · have h_closed : IsClosed { x | ¬Surjective (fderiv ℝ (L ∘ f) x) } := by
      apply_rules [ isClosed_criticalSet ] ; exact hf.comp ( contDiff_id ) |> ContDiff.comp ( L.contDiff ) ;
    generalize_proofs at *; (
    have h_countable_union : L ∘ f '' { x | ¬Surjective (fderiv ℝ (L ∘ f) x) } = ⋃ n : ℕ, L ∘ f '' (Metric.closedBall 0 (n : ℝ) ∩ { x | ¬Surjective (fderiv ℝ (L ∘ f) x) }) := by
      ext x
      simp [Set.mem_iUnion, Set.mem_image];
      exact ⟨ fun ⟨ y, hy, hy' ⟩ => ⟨ ⌈‖y‖⌉₊, y, ⟨ Nat.le_ceil _, hy ⟩, hy' ⟩, fun ⟨ n, y, ⟨ hy, hy' ⟩, hy'' ⟩ => ⟨ y, hy', hy'' ⟩ ⟩
    generalize_proofs at *; (
    rw [h_countable_union];
    refine' MeasurableSet.iUnion fun n => _;
    exact IsCompact.measurableSet ( IsCompact.image ( ProperSpace.isCompact_closedBall _ _ |> IsCompact.inter_right <| h_closed ) <| Continuous.comp ( L.continuous ) <| hf.continuous )))))); -- To preserve the `generalize_proofs` context. (This line can be removed after conversion to `generalize_proofs at *` at `all_goals`.)

end Reduction

/-! ### High-dimensional case: finrank E > finrank F -/

section HighDimensional

variable {E F : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E] [FiniteDimensional ℝ E]
  [NormedAddCommGroup F] [NormedSpace ℝ F] [FiniteDimensional ℝ F]
  [MeasurableSpace E] [BorelSpace E]
  [MeasurableSpace F] [BorelSpace F]

/-- **Sard's theorem** when `finrank E > finrank F`.

This is the hardest case of Sard's theorem. The standard proof proceeds by strong
induction on `finrank E`, decomposing the critical set into:

* **Shallow critical points** (where `df ≠ 0`): The implicit function theorem provides
  a local change of coordinates reducing to a lower-dimensional Sard problem.

* **Deep critical points** (where all derivatives up to order `K` vanish): Taylor's
  theorem with `K ≥ ⌈finrank E / finrank F⌉` shows the image has small covering
  numbers, hence measure zero.

The formalization of this inductive argument requires substantial infrastructure
(multivariate Taylor remainder estimates, the implicit function theorem in coordinate
form, careful measurability arguments for the stratified critical set) and is
left for future work. -/
theorem sard_of_finrank_gt
    (μ : Measure F) [μ.IsAddHaarMeasure]
    (f : E → F) (hf : ContDiff ℝ ⊤ f)
    (hdim : Module.finrank ℝ F < Module.finrank ℝ E) :
    μ (f '' {x | ¬ Surjective (fderiv ℝ f x)}) = 0 := by
  sorry

end HighDimensional

/-! ### General Sard's Theorem -/

section General

/-- **Sard's Theorem**: For a C^∞ smooth map `f` between finite-dimensional real
normed spaces, the set of critical values (the image of points where the derivative
is not surjective) has Lebesgue measure zero.

This is one of the fundamental results of differential topology, with applications
to transversality theory, Morse theory, and the study of generic properties of
smooth maps.

The proof reduces to three cases:
* `finrank E < finrank F`: handled by `addHaar_image_eq_zero_of_finrank_lt`
* `finrank E = finrank F`: handled by `sard_of_finrank_eq` via the area formula
* `finrank E > finrank F`: handled by `sard_of_finrank_gt` via induction -/
theorem sard
    {E F : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E] [FiniteDimensional ℝ E]
    [NormedAddCommGroup F] [NormedSpace ℝ F] [FiniteDimensional ℝ F]
    [MeasurableSpace E] [BorelSpace E]
    [MeasurableSpace F] [BorelSpace F]
    (μ : Measure F) [μ.IsAddHaarMeasure]
    (f : E → F) (hf : ContDiff ℝ ⊤ f) :
    μ (f '' {x | ¬ Surjective (fderiv ℝ f x)}) = 0 := by
  rcases lt_trichotomy (Module.finrank ℝ E) (Module.finrank ℝ F) with h | h | h
  · -- Case: finrank E < finrank F.
    exact le_antisymm
      ((measure_mono (image_subset_range f _)).trans
        (addHaar_range_eq_zero_of_finrank_lt μ f (hf.of_le le_top) h).le)
      bot_le
  · -- Case: finrank E = finrank F.
    exact sard_of_finrank_eq μ f (hf.of_le le_top) h
  · -- Case: finrank E > finrank F.
    exact sard_of_finrank_gt μ f hf h

end General
