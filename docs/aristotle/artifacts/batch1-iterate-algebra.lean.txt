import Mathlib

open Function

variable {X : Type*}

/-
PROBLEM
Injective f implies injective f^[n].

PROVIDED SOLUTION
Use Function.Injective.iterate from Mathlib if available, otherwise induct on n. Base case: id is injective. Inductive step: f^[n+1] = f ∘ f^[n], composition of injectives is injective.
-/
theorem injective_iterate' (f : X → X) (hf : Injective f) (n : ℕ) :
    Injective (f^[n]) := by
  exact hf.iterate n

/-
PROBLEM
If f^[n] x = f^[n] y for all n < k, and k > 0, then x = y.

PROVIDED SOLUTION
Use the hypothesis h with n=0 (which satisfies n < k since k > 0). Then f^[0] x = f^[0] y simplifies to x = y.
-/
theorem eq_of_iterate_eq_range (f : X → X) (x y : X) (k : ℕ)
    (hk : 0 < k)
    (h : ∀ n, n < k → f^[n] x = f^[n] y) :
    x = y := by
  simpa using h 0 hk

/-- Orbit of x under f as a Finset, bounded by n steps. -/
def orbitFinset [DecidableEq X] (f : X → X) (x : X) (n : ℕ) :
    Finset X :=
  (Finset.range n).image (fun i => f^[i] x)

/-
PROBLEM
Orbit cardinality is at most n.

PROVIDED SOLUTION
orbitFinset is defined as (Finset.range n).image (...). Use Finset.card_image_le to get card of image ≤ card of source, and Finset.card_range to get card (Finset.range n) = n.
-/
theorem card_orbitFinset_le [DecidableEq X] (f : X → X) (x : X)
    (n : ℕ) :
    (orbitFinset f x n).card ≤ n := by
  exact le_trans ( Finset.card_image_le ) ( by simp )

/-
PROBLEM
On finite type, orbit hits every element or repeats.

PROVIDED SOLUTION
orbitFinset f x n is a Finset X. On a Fintype X, any Finset X has card ≤ Fintype.card X. Use Finset.card_le_univ.
-/
theorem orbitFinset_card_le_card [Fintype X] [DecidableEq X]
    (f : X → X) (x : X) (n : ℕ) :
    (orbitFinset f x n).card ≤ Fintype.card X := by
  exact Finset.card_le_univ _
