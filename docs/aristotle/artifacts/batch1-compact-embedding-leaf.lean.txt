/-
Aristotle Batch 1J — Compact space embedding lemmas
Target: the general compact → T2 embedding theorem
-/
import Mathlib

open Function Set Topology

variable {X Y : Type*} [TopologicalSpace X] [TopologicalSpace Y]

/-- Continuous bijection from compact to T2 is a homeomorphism. -/
theorem isHomeomorph_of_continuous_bijective_compact_t2
    [CompactSpace X] [T2Space Y]
    {f : X → Y} (hf : Continuous f) (hbij : Bijective f) :
    IsHomeomorph f := by
  refine' ⟨hf, _, _⟩
  · intro U hU
    have h_closed : IsClosed (f '' Uᶜ) :=
      IsCompact.isClosed (IsCompact.image (isClosed_compl_iff.mpr hU |> IsClosed.isCompact) hf)
    have h_compl : f '' U = (f '' Uᶜ)ᶜ := by
      ext y; simp [Set.mem_image, Set.mem_compl_iff]
      exact ⟨fun ⟨x, hx, hy⟩ z hz hz' =>
        hz (hbij.injective (hz'.trans hy.symm) ▸ hx),
        fun h => by rcases hbij.surjective y with ⟨x, rfl⟩;
                    exact ⟨x, by contrapose! h; tauto, rfl⟩⟩
    rw [h_compl]
    exact isOpen_compl_iff.mpr h_closed
  · exact hbij

/-- Continuous injection from compact to T2 is an embedding. -/
theorem isEmbedding_of_continuous_injective_compact_t2
    [CompactSpace X] [T2Space Y]
    {f : X → Y} (hf : Continuous f) (hinj : Injective f) :
    IsEmbedding f :=
  (Continuous.isClosedEmbedding hf hinj).isEmbedding

/-- Continuous injection from compact to T2 is a closed embedding. -/
theorem isClosedEmbedding_of_continuous_injective_compact_t2
    [CompactSpace X] [T2Space Y]
    {f : X → Y} (hf : Continuous f) (hinj : Injective f) :
    IsClosedEmbedding f :=
  Continuous.isClosedEmbedding hf hinj
