import Mathlib

open Function

variable {X : Type*}

def delayMap (f : X → X) (α : X → ℝ) (k : ℕ) (x : X) : Fin k → ℝ :=
  fun i => α (f^[i.val] x)

/-
PROBLEM
Delay map shift: applying f shifts the window by one step.

PROVIDED SOLUTION
Unfold delayMap, then rewrite using Function.iterate_succ_apply' to show f^[i+1] x = f^[i] (f x).
-/
theorem delayMap_comp_eq (f : X → X) (α : X → ℝ) (k : ℕ) (x : X)
    (i : Fin k) :
    delayMap f α k (f x) i = α (f^[i.val + 1] x) := by
  -- By definition of $delayMap$, we have $delayMap f α k (f x) i = α (f^[i.val] (f x))$.
  simp [delayMap]

/-
PROBLEM
HELD: delayMap_injective_of_injective_f needs mathematical review.
The claim that injective f + injective k-window → injective (k+1)-window
is stronger than separatesOrbits_of_le and may be false.
Do not submit until the statement is verified.

If α ∘ f^[i] = β ∘ f^[i] for all i < k, the delay maps agree.

PROVIDED SOLUTION
Unfold delayMap, apply funext, then use the hypothesis h for each index i.
-/
theorem delayMap_eq_of_comp_iterate_eq (f : X → X) (α β : X → ℝ)
    (k : ℕ) (x : X)
    (h : ∀ i : Fin k, α (f^[i.val] x) = β (f^[i.val] x)) :
    delayMap f α k x = delayMap f β k x := by
  exact funext fun i => h i
