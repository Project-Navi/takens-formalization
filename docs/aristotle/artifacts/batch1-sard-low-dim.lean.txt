import Mathlib

open MeasureTheory Set Function FiniteDimensional

noncomputable section

variable {E F : Type*} [NormedAddCommGroup E] [NormedSpace ℝ E]
  [FiniteDimensional ℝ E] [NormedAddCommGroup F] [NormedSpace ℝ F]
  [FiniteDimensional ℝ F] [MeasurableSpace E] [BorelSpace E]
  [MeasurableSpace F] [BorelSpace F]

omit [MeasurableSpace E] [BorelSpace E] in
/-- Image of any set under C¹ map from lower-dim space has measure zero. -/
theorem addHaar_image_eq_zero_of_finrank_lt
    (μ : Measure F) [μ.IsAddHaarMeasure]
    (f : E → F) (hf : ContDiff ℝ 1 f) (s : Set E)
    (hdim : Module.finrank ℝ E < Module.finrank ℝ F) :
    μ (f '' s) = 0 := by
  have h_dim_le : dimH (f '' s) ≤ (Module.finrank ℝ E) := by
    have h_lip : LocallyLipschitz f := hf.locallyLipschitz
    have h_dim_image : ∀ x ∈ s, ∃ C : NNReal, ∃ t ∈ nhdsWithin x s, LipschitzOnWith C f t := by
      intro x hx
      obtain ⟨ C, t, ht, hC ⟩ := h_lip x
      exact ⟨ C, t, mem_nhdsWithin_of_mem_nhds ht, hC ⟩
    refine' le_trans ( dimH_image_le_of_locally_lipschitzOn _ ) _
    · exact h_dim_image
    · have h_dim_s : dimH (Set.univ : Set E) = Module.finrank ℝ E :=
        Real.dimH_univ_eq_finrank E
      exact h_dim_s ▸ dimH_mono ( Set.subset_univ s )
  have h_abs_cont : μ.AbsolutelyContinuous (MeasureTheory.Measure.hausdorffMeasure (Module.finrank ℝ F)) :=
    Measure.absolutelyContinuous_isAddHaarMeasure μ _
  refine' h_abs_cont _
  refine' MeasureTheory.Measure.hausdorffMeasure_zero_or_top _ _ |> Or.resolve_right <| _
  exact ↑ ( Module.finrank ℝ E ) + ( ↑ ( Module.finrank ℝ F ) - ↑ ( Module.finrank ℝ E ) ) / 2
  · linarith [ show ( Module.finrank ℝ E : ℝ ) < Module.finrank ℝ F by exact_mod_cast hdim ]
  · contrapose! h_dim_le
    refine' lt_of_lt_of_le _ ( le_dimH_of_hausdorffMeasure_ne_zero _ )
    rotate_left
    exact ⟨ Module.finrank ℝ E + ( Module.finrank ℝ F - Module.finrank ℝ E ) / 2, by exact add_nonneg ( Nat.cast_nonneg _ ) ( div_nonneg ( sub_nonneg.mpr ( Nat.cast_le.mpr hdim.le ) ) zero_le_two ) ⟩
    · aesop
    · nontriviality
      erw [ ENNReal.coe_lt_coe ]
      exact Subtype.mk_lt_mk.mpr ( lt_add_of_pos_right _ ( div_pos ( sub_pos.mpr ( Nat.cast_lt.mpr hdim ) ) zero_lt_two ) )

omit [MeasurableSpace E] [BorelSpace E] in
/-- Range of C¹ map from lower-dim space has measure zero. -/
theorem addHaar_range_eq_zero_of_finrank_lt
    (μ : Measure F) [μ.IsAddHaarMeasure]
    (f : E → F) (hf : ContDiff ℝ 1 f)
    (hdim : Module.finrank ℝ E < Module.finrank ℝ F) :
    μ (range f) = 0 := by
  have := addHaar_image_eq_zero_of_finrank_lt μ f hf Set.univ hdim
  aesop
