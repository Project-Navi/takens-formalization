import Mathlib

/-!
# Delay Embeddings for Finite Dynamical Systems

We formalize the theory of delay embeddings for finite dynamical systems `(X, f)` where
`X` is a finite type and `f : X → X`. We characterize exactly when a (possibly non-injective)
observation `α : X → ℝ` gives an injective delay embedding, relate the minimum window length
to coincidence lengths between distinct orbit pairs, and prove that for permutations,
`k = card X` suffices for any injective observation.

## Main Definitions

* `DelayEmbedding.delayEmbedding` - The delay embedding map `x ↦ (α(x), α(f(x)), ..., α(f^k(x)))`
* `DelayEmbedding.SeparatesOrbits` - Predicate that the delay embedding is injective
* `DelayEmbedding.coincidenceLength` - The first iterate where orbits of two points differ under α

## Main Results

* `DelayEmbedding.separatesOrbits_iff` - Characterization: the delay embedding is injective iff
  for all distinct pairs, their orbits differ in α-value within k steps
* `DelayEmbedding.separatesOrbits_mono` - Monotonicity in the window length
* `DelayEmbedding.separatesOrbits_of_injective` - An injective observation separates at any window
* `DelayEmbedding.exists_separating_window_iff` - A (possibly non-injective) α can give an
  injective delay embedding iff for every distinct pair, orbits eventually differ under α
* `DelayEmbedding.separatesOrbits_iff_coincidence` - Separation at window k ↔ all coincidence
  lengths are at most k
* `DelayEmbedding.separatesOrbits_card_of_bijective` - For bijective f and injective α,
  window length `card X` suffices
-/

namespace DelayEmbedding

open Function Fintype Classical
set_option linter.unusedSectionVars false

variable {X : Type*} [Fintype X] [DecidableEq X]

/-- The delay embedding maps a point `x` to the tuple
    `(α(x), α(f(x)), α(f²(x)), ..., α(fᵏ(x)))` as a function `Fin (k+1) → ℝ`. -/
def delayEmbedding (f : X → X) (α : X → ℝ) (k : ℕ) (x : X) : Fin (k + 1) → ℝ :=
  fun i => α (f^[i.val] x)

/-- `SeparatesOrbits f α k` means the delay embedding with observation `α` and
    window length `k` is injective: distinct points yield distinct observation tuples. -/
def SeparatesOrbits (f : X → X) (α : X → ℝ) (k : ℕ) : Prop :=
  Injective (delayEmbedding f α k)

/-! ### Characterization of injective delay embeddings -/

/-- The delay embedding is injective iff for all distinct `x, y ∈ X`,
    there exists some `i ≤ k` such that `α(fⁱ(x)) ≠ α(fⁱ(y))`. -/
theorem separatesOrbits_iff (f : X → X) (α : X → ℝ) (k : ℕ) :
    SeparatesOrbits f α k ↔
    ∀ x y, x ≠ y → ∃ i : Fin (k + 1), α (f^[i.val] x) ≠ α (f^[i.val] y) := by
  unfold SeparatesOrbits
  constructor
  · intro h x y hxy
    exact not_forall.mp fun H => hxy (h <| funext fun i => H i)
  · intro h x y hxy
    exact not_not.1 fun hne => by obtain ⟨i, hi⟩ := h x y hne; exact hi (congr_fun hxy i)

/-! ### Monotonicity -/

/-- If `α` separates orbits at window `k`, it also separates at any larger window `k' ≥ k`. -/
theorem separatesOrbits_mono {f : X → X} {α : X → ℝ} {k k' : ℕ} (hk : k ≤ k')
    (h : SeparatesOrbits f α k) : SeparatesOrbits f α k' := by
  intro x y hxy
  apply h
  exact funext fun i => congr_fun hxy ⟨i, by linarith [Fin.is_lt i]⟩

/-! ### Injective observations -/

/-- If `α` is injective, then it separates orbits at *any* window length,
    including `k = 0`. This holds regardless of the dynamics `f`. -/
theorem separatesOrbits_of_injective (f : X → X) {α : X → ℝ} (hα : Injective α) (k : ℕ) :
    SeparatesOrbits f α k :=
  fun _ _ hxy => hα (by simpa [Function.iterate_zero] using congr_fun hxy 0)

/-! ### Non-injective observations: when do they still work? -/

/-- A (possibly non-injective) observation `α` can give an injective delay embedding
    (for some window length) if and only if for every pair of distinct points,
    their orbits eventually produce different `α`-values. This characterizes exactly
    when a non-injective `α` still yields an injective delay embedding. -/
theorem exists_separating_window_iff (f : X → X) (α : X → ℝ) :
    (∃ k, SeparatesOrbits f α k) ↔
    ∀ x y, x ≠ y → ∃ i : ℕ, α (f^[i] x) ≠ α (f^[i] y) := by
  constructor <;> intro h
  · intro x y hxy
    rcases h with ⟨k, hk⟩
    rcases (separatesOrbits_iff f α k).mp hk x y hxy with ⟨i, hi⟩
    exact ⟨i, hi⟩
  · choose! i hi using h
    obtain ⟨k, hk⟩ : ∃ k, ∀ x y, x ≠ y → i x y ≤ k :=
      ⟨Finset.univ.sup fun x => Finset.univ.sup fun y => i x y,
       fun x y _ => Finset.le_sup_of_le (Finset.mem_univ x)
         (Finset.le_sup_of_le (Finset.mem_univ y) le_rfl)⟩
    use k
    intro x y hxy
    exact not_not.1 fun h => hi x y h <|
      congr_fun hxy ⟨i x y, Nat.lt_succ_of_le (hk x y h)⟩

/-! ### Coincidence lengths and the minimum window -/

/-- The coincidence length of two points `x` and `y` under dynamics `f` and
    observation `α`: the index of the first iterate where `α(fⁱ(x)) ≠ α(fⁱ(y))`,
    or `⊤` (infinity) if their orbits always agree under `α`. -/
noncomputable def coincidenceLength (f : X → X) (α : X → ℝ) (x y : X) : ℕ∞ :=
  if h : ∃ i : ℕ, α (f^[i] x) ≠ α (f^[i] y)
  then ↑(Nat.find h)
  else ⊤

/-- The coincidence length is zero when `α` distinguishes `x` and `y` directly. -/
theorem coincidenceLength_eq_zero {f : X → X} {α : X → ℝ} {x y : X}
    (h : α x ≠ α y) : coincidenceLength f α x y = 0 := by
  unfold coincidenceLength
  split_ifs with h'
  · simp +zetaDelta at *; exact h
  · exact False.elim (h' ⟨0, by simpa using h⟩)

/-- The delay embedding separates orbits at window `k` if and only if the
    coincidence length of every distinct pair is at most `k`.
    This establishes the precise relationship: the minimum window for separation
    equals the maximum coincidence length over all distinct pairs. -/
theorem separatesOrbits_iff_coincidence (f : X → X) (α : X → ℝ) (k : ℕ) :
    SeparatesOrbits f α k ↔
    ∀ x y, x ≠ y → coincidenceLength f α x y ≤ ↑k := by
  constructor <;> intro H x y hxy
  · unfold coincidenceLength
    split_ifs with h <;> simp_all +decide [SeparatesOrbits]
    · contrapose! H
      exact fun h => hxy <| h <| funext fun i => H i.val i.is_le
    · exact hxy (H <| funext fun i => h i.val)
  · contrapose! H
    refine ⟨x, y, H, ?_⟩
    unfold coincidenceLength
    split_ifs <;> simp_all +decide [funext_iff, delayEmbedding]
    exact fun m hm => hxy ⟨m, Nat.lt_succ_of_le hm⟩

/-! ### Permutations (bijective dynamics) -/

/-- **Permutation embedding theorem**: If `f` is a bijection (permutation) on a finite type
    and `α` is an injective observation, then `k = card X` suffices for the delay embedding
    to be injective. (In fact, any `k` suffices when `α` is injective, but this gives the
    explicit bound matching the user's request.) -/
theorem separatesOrbits_card_of_bijective {f : X → X} (_hf : Bijective f)
    {α : X → ℝ} (hα : Injective α) :
    SeparatesOrbits f α (Fintype.card X) :=
  separatesOrbits_of_injective f hα _

end DelayEmbedding
