/-
Aristotle Batch 1L — Finite separation theory
Target: when does a finite observation separate a finite state space?
-/
import Mathlib

set_option linter.unusedSectionVars false

open Function Fintype

variable {X : Type*} [Fintype X] [DecidableEq X]

/-- An observation α : X → ℝ separates points of X iff it is injective. -/
theorem separates_iff_injective (α : X → ℝ) :
    (∀ x y : X, α x = α y → x = y) ↔ Injective α :=
  ⟨fun a _ _ a_1 => a _ _ a_1, fun a _ _ a_1 => a a_1⟩

/-- For finite X, the number of distinct values of α is at most card X. -/
theorem card_range_le (α : X → ℝ) :
    (Finset.univ.image α).card ≤ Fintype.card X := by
  convert Finset.card_image_le

/-- If α has card X distinct values on a finite type, α is injective. -/
theorem injective_of_card_image_eq (α : X → ℝ)
    (h : (Finset.univ.image α).card = Fintype.card X) :
    Injective α := by
  have := Finset.card_image_iff.mp
    (by aesop : Finset.card (Finset.image α Finset.univ) =
      Finset.card (Finset.univ : Finset X))
  aesop

/-- The minimum window length for orbit separation depends on the
    maximum coincidence length between orbit pairs. -/
theorem separatesOrbits_of_no_coincidence
    (f : X → X) (α : X → ℝ) (k : ℕ)
    (h : ∀ x y : X, x ≠ y →
      ∃ i : Fin k, α (f^[i.val] x) ≠ α (f^[i.val] y)) :
    ∀ x y : X, (∀ i : Fin k, α (f^[i.val] x) = α (f^[i.val] y)) →
      x = y := by
  intro x y hxy; contrapose! h; aesop
