/-
Aristotle Batch 1F — Orbit separation depth
Target: connect SeparatesOrbits to concrete finite-type conditions
-/
import Mathlib

open Function Fintype

variable {X : Type*}

def SeparatesOrbits' (f : X → X) (α : X → ℝ) (k : ℕ) : Prop :=
  ∀ x y : X, (∀ i : Fin k, α (f^[i.val] x) = α (f^[i.val] y)) → x = y

/-
PROBLEM
REMOVED: separatesOrbits_mono duplicates separatesOrbits_of_le
from DelayWindow.lean. Use the canonical version.

On a finite type, the number of distinct delay windows is at most
    card X.

PROVIDED SOLUTION
Use Finset.card_image_le to get card (image ...) ≤ card univ, then Finset.card_univ to rewrite card univ = Fintype.card X.
-/
theorem delay_image_card_le [Fintype X] [DecidableEq X]
    (f : X → X) (α : X → ℝ) (k : ℕ) :
    (Finset.univ.image (fun x => fun i : Fin k => α (f^[i.val] x))).card
      ≤ Fintype.card X := by
  convert Finset.card_image_le

/-
PROBLEM
Injective delay map on finite type → card bound on windows.

PROVIDED SOLUTION
Use Finset.card_image_of_injective with the injectivity hypothesis to rewrite (univ.image f).card = univ.card, then use Finset.card_univ.
-/
theorem card_le_of_injective_delay [Fintype X] [DecidableEq X]
    (f : X → X) (α : X → ℝ) (k : ℕ)
    (h : Injective (fun x => fun i : Fin k => α (f^[i.val] x))) :
    Fintype.card X ≤
      (Finset.univ.image (fun x => fun i : Fin k => α (f^[i.val] x))).card := by
  rw [ Finset.card_image_of_injective _ h, Finset.card_univ ]
