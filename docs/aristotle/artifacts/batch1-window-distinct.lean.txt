/-
Aristotle Batch 1K — WindowDistinct conditions
Target: when are delay window values distinct?
-/
import Mathlib

open Function

variable {X : Type*}

def delayWin (f : X → X) (α : X → ℝ) (k : ℕ) (x : X) : Fin k → ℝ :=
  fun i => α (f^[i.val] x)

/-
PROBLEM
If α is injective and the orbit doesn't repeat within k steps,
    the delay window has distinct values.

PROVIDED SOLUTION
Unfold delayWin. The function is fun i => α (f^[i.val] x). To show injectivity, take i j with α(f^[i] x) = α(f^[j] x). By injectivity of α, f^[i] x = f^[j] x. By h, i = j.
-/
theorem windowDistinct_of_injective_orbit
    (f : X → X) (α : X → ℝ) (hα : Injective α) (k : ℕ) (x : X)
    (h : ∀ i j : Fin k, f^[i.val] x = f^[j.val] x → i = j) :
    Injective (delayWin f α k x) := by
  intro i j; aesop;

/-
PROBLEM
If f is injective, the orbit doesn't repeat (no preperiod).

PROVIDED SOLUTION
By cases on i = j or i ≠ j. If i = j, Left. If i ≠ j, use Nat.lt_or_gt_of_ne. Case i < j: let p = j - i, so p > 0 and j = i + p. Then f^[i+p] x = f^[i] x. Use iterate_add_apply to get f^[i](f^[p] x) = f^[i] x. By (hf.iterate i), f^[p] x = x. Right ⟨p, by omega, Or.inr rfl, this⟩. Case i > j is symmetric with p = i - j.
-/
theorem orbit_injective_of_injective
    (f : X → X) (hf : Injective f) (x : X) (i j : ℕ)
    (h : f^[i] x = f^[j] x) : i = j ∨ ∃ p, 0 < p ∧
      (i = j + p ∨ j = i + p) ∧ f^[p] x = x := by
  cases' lt_trichotomy i j with hij hij;
  · refine' Or.inr ⟨ j - i, tsub_pos_of_lt hij, Or.inr ( Eq.symm ( add_tsub_cancel_of_le hij.le ) ), _ ⟩;
    exact hf.iterate i ( by simpa [ ← Function.iterate_add_apply, hij.le ] using h.symm );
  · cases' hij with hij hij;
    · exact Or.inl hij;
    · -- Since $i > j$, we can write $i = j + p$ for some $p > 0$.
      obtain ⟨p, hp⟩ : ∃ p, i = j + p ∧ 0 < p := by
        exact ⟨ i - j, by rw [ Nat.add_sub_cancel' hij.le ], Nat.sub_pos_of_lt hij ⟩;
      simp_all +decide [ Function.iterate_add_apply ];
      exact Or.inr ⟨ p, hp.2, Or.inl rfl, hf.iterate j h ⟩
