import Mathlib

/-!
# Closed embeddings from compact to Hausdorff spaces, and the delay coordinate map

## Main results

1. `compactToT2_isClosedEmbedding`: A continuous injection from a compact space to a
   Hausdorff space is a closed embedding.
2. `compactToT2_rangeHomeomorph`: The range-factorization of such a map is a homeomorphism.
3. `delayCoordinateMap_continuous`: The delay coordinate map `fun x i => h (T^[i] x)` is
   continuous when `T` and `h` are continuous.
4. `delayCoordinateMap_isClosedEmbedding`: If the delay coordinate map is injective on a
   compact space mapping to `Fin n → ℝ` (which is T₂), then it is a closed embedding.
5. `delayCoordinateMap_rangeHomeomorph`: The range factorization of the delay coordinate
   map is a homeomorphism under the same hypotheses.
-/

open Function Topology Set

noncomputable section

/-! ## General theory: compact → T₂ continuous injections -/

/-- A continuous injection from a compact space to a Hausdorff space is a closed embedding. -/
theorem compactToT2_isClosedEmbedding
    {X Y : Type*} [TopologicalSpace X] [TopologicalSpace Y]
    [CompactSpace X] [T2Space Y]
    {f : X → Y} (hf : Continuous f) (hinj : Injective f) :
    IsClosedEmbedding f :=
  hf.isClosedEmbedding hinj

/-- The range factorization of a continuous injection from a compact space to a Hausdorff
space is a homeomorphism onto the range. -/
def compactToT2_rangeHomeomorph
    {X Y : Type*} [TopologicalSpace X] [TopologicalSpace Y]
    [CompactSpace X] [T2Space Y]
    {f : X → Y} (hf : Continuous f) (hinj : Injective f) :
    X ≃ₜ range f := by
  have he : IsClosedEmbedding f := hf.isClosedEmbedding hinj
  refine (Equiv.ofInjective f hinj).toHomeomorphOfIsInducing ?_
  have : IsInducing (rangeFactorization f) :=
    he.isEmbedding.isInducing.codRestrict (mem_range_self)
  convert this using 1

/-! ## Delay coordinate map -/

/-- The delay coordinate map sends `x` to the tuple `(h(x), h(T(x)), …, h(T^{n-1}(x)))`. -/
def delayCoordinateMap {X : Type*} (T : X → X) (h : X → ℝ) (n : ℕ) : X → (Fin n → ℝ) :=
  fun x i => h (T^[i] x)

/-- The delay coordinate map is continuous when both `T` and `h` are continuous. -/
theorem delayCoordinateMap_continuous
    {X : Type*} [TopologicalSpace X]
    {T : X → X} {h : X → ℝ} {n : ℕ}
    (hT : Continuous T) (hh : Continuous h) :
    Continuous (delayCoordinateMap T h n) := by
  apply continuous_pi
  intro i
  exact hh.comp (hT.iterate i)

/-- The delay coordinate map from a compact space is a closed embedding when it is injective. -/
theorem delayCoordinateMap_isClosedEmbedding
    {X : Type*} [TopologicalSpace X] [CompactSpace X]
    {T : X → X} {h : X → ℝ} {n : ℕ}
    (hT : Continuous T) (hh : Continuous h)
    (hinj : Injective (delayCoordinateMap T h n)) :
    IsClosedEmbedding (delayCoordinateMap T h n) :=
  compactToT2_isClosedEmbedding (delayCoordinateMap_continuous hT hh) hinj

/-- When the delay coordinate map is injective on a compact space, its range factorization
is a homeomorphism onto its image in `Fin n → ℝ`. -/
def delayCoordinateMap_rangeHomeomorph
    {X : Type*} [TopologicalSpace X] [CompactSpace X]
    {T : X → X} {h : X → ℝ} {n : ℕ}
    (hT : Continuous T) (hh : Continuous h)
    (hinj : Injective (delayCoordinateMap T h n)) :
    X ≃ₜ range (delayCoordinateMap T h n) :=
  compactToT2_rangeHomeomorph (delayCoordinateMap_continuous hT hh) hinj

end
