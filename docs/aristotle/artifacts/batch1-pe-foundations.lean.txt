/-
Aristotle Batch 1H — Permutation entropy foundations
Target: ordinal pattern distribution on finite orbits
-/
import Mathlib

open Function Fintype Finset

noncomputable section

variable {X : Type*} [Fintype X] [DecidableEq X]

/-- The set of ordinal patterns observed along an orbit of length N
    with window size d. -/
def observedPatterns (f : X → X) (α : X → ℝ) (d : ℕ)
    (x : X) (N : ℕ) : Finset (Equiv.Perm (Fin d)) :=
  (Finset.range N).image (fun t =>
    Tuple.sort (fun i : Fin d => α (f^[t + i.val] x)))

/-
PROBLEM
The number of observed patterns is at most d!.

PROVIDED SOLUTION
The observedPatterns is a Finset of Equiv.Perm (Fin d). It's a subset of Finset.univ. Use Finset.card_le_card (Finset.subset_univ _), then rewrite Finset.card_univ and Fintype.card_perm.
-/
theorem card_observedPatterns_le (f : X → X) (α : X → ℝ) (d : ℕ)
    (x : X) (N : ℕ) :
    (observedPatterns f α d x N).card ≤ d.factorial := by
  convert Finset.card_le_card _ using 1;
  rotate_left;
  exact Finset.univ.image fun σ : Equiv.Perm ( Fin d ) => σ;
  · intro σ hσ
    aesop;
  · simp +decide [ Fintype.card_perm ]

/-
PROBLEM
The number of observed patterns is at most N.

PROVIDED SOLUTION
Unfold observedPatterns, then apply Finset.card_image_le which gives (Finset.image f s).card ≤ s.card, then use Finset.card_range N.
-/
theorem card_observedPatterns_le_N (f : X → X) (α : X → ℝ) (d : ℕ)
    (x : X) (N : ℕ) :
    (observedPatterns f α d x N).card ≤ N := by
  exact le_trans ( Finset.card_image_le ) ( by simp +decide )

/-
PROBLEM
The original statement below is false when `x` is not periodic, since
   `minimalPeriod f x = 0` in that case while the pattern set can be nonempty.
   We comment it out and provide a corrected version with a periodicity hypothesis.

theorem card_observedPatterns_le_period (f : X → X) (α : X → ℝ)
    (d : ℕ) (x : X) (N : ℕ) :
    (observedPatterns f α d x N).card ≤
      minimalPeriod f x := by
  sorry

Helper: on a periodic orbit, iterates at congruent times agree.

PROVIDED SOLUTION
Write t = (minimalPeriod f x) * q + r where r = t % minimalPeriod f x. Then f^[t] x = f^[p*q + r] x. Since f^[p] x = x (where p = minimalPeriod f x), by induction on q we get f^[p*q + r] x = f^[r] x. Use Function.IsPeriodicPt.iterate_mod_apply or similar Mathlib lemmas about iterate and minimalPeriod.
-/
lemma iterate_mod_minimalPeriod (f : X → X) (x : X)
    (hx : x ∈ periodicPts f) (t : ℕ) :
    f^[t] x = f^[t % minimalPeriod f x] x := by
  induction t / minimalPeriod f x <;> simp_all +decide [ Nat.mul_succ, ← add_assoc ]

/-
PROBLEM
Corrected version: on a periodic orbit of period p, the number of
    distinct ordinal patterns is at most p.

PROVIDED SOLUTION
Unfold observedPatterns. The key insight: for periodic x with period p, f^[t+i] x = f^[(t+i) % p] x by iterate_mod_minimalPeriod. So the pattern at time t equals the pattern at time (t % p). Thus the image of range N factors through range p: the set of patterns from range N is a subset of the set of patterns from range p. Formally:
1. Show the image over range N is a subset of the image over range p (using that t % p ∈ range p and the patterns agree).
2. Apply card_le_card to get card ≤ card of image over range p.
3. Apply card_image_le to get card of image over range p ≤ p.
-/
theorem card_observedPatterns_le_period' (f : X → X) (α : X → ℝ)
    (d : ℕ) (x : X) (N : ℕ) (hx : x ∈ periodicPts f) :
    (observedPatterns f α d x N).card ≤ minimalPeriod f x := by
  -- The image of range N under the pattern function is a subset of the image of range p under the pattern function, by the lemma `iterate_mod_minimalPeriod`.
  have hsubset : (observedPatterns f α d x N) ⊆ Finset.image (fun t => (Tuple.sort (fun i : Fin d => α (f^[t + i.val] x)))) (Finset.range (minimalPeriod f x)) := by
    intro t ht
    obtain ⟨t', ht', rfl⟩ := Finset.mem_image.mp ht
    have ht'_mod : t' % minimalPeriod f x < minimalPeriod f x := by
      exact Nat.mod_lt _ ( Function.minimalPeriod_pos_of_mem_periodicPts hx )
    have ht'_eq : t' % minimalPeriod f x ∈ Finset.range (minimalPeriod f x) := by
      exact Finset.mem_range.mpr ht'_mod;
    refine' Finset.mem_image.mpr ⟨ t' % minimalPeriod f x, ht'_eq, _ ⟩;
    have ht'_eq : ∀ i : Fin d, f^[t' + i.val] x = f^[t' % minimalPeriod f x + i.val] x := by
      intro i;
      rw [ ← Nat.mod_add_div t' ( minimalPeriod f x ), add_right_comm ];
      induction t' / minimalPeriod f x <;> simp_all +decide [ Nat.mul_succ, ← add_assoc, Function.iterate_add_apply ];
    simp +decide only [ht'_eq];
  exact le_trans ( Finset.card_le_card hsubset ) ( Finset.card_image_le.trans ( by simp ) )
